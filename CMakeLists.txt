cmake_minimum_required(VERSION 3.24)
project(QuickQR VERSION 1.0.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ZXING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZXING_C_API OFF CACHE BOOL "" FORCE)
set(ZXING_WRITERS OLD CACHE STRING "" FORCE)
set(ZXING_LINK_CPP_STATICALLY ON CACHE BOOL "" FORCE)
foreach(format 1D AZTEC DATAMATRIX MAXICODE PDF417)
  set(ZXING_ENABLE_${format} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(third_party/zxing-cpp-3.1.1 EXCLUDE_FROM_ALL)
add_library(qr_core STATIC core.cpp)
target_link_libraries(qr_core PUBLIC ZXing::ZXing gdiplus)
target_compile_definitions(qr_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(qr_core PRIVATE /utf-8 /W4 /O1 /Gy)
add_executable(QuickQR WIN32 main.cpp app.rc)
target_link_libraries(QuickQR PRIVATE qr_core comctl32 shell32 shlwapi ole32 dwmapi comdlg32)
target_compile_options(QuickQR PRIVATE /utf-8 /W4 /O1 /Gy)
target_link_options(QuickQR PRIVATE /OPT:REF /OPT:ICF /MANIFEST:NO)
add_executable(qr_tests tests/tests.cpp)
target_link_libraries(qr_tests PRIVATE qr_core)
target_include_directories(qr_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(qr_tests PRIVATE /utf-8 /W4)
enable_testing()
add_test(NAME qr_regression COMMAND qr_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures)

add_executable(qr_ui_tests WIN32 EXCLUDE_FROM_ALL tests/ui_tests.cpp app.rc)
target_compile_definitions(qr_ui_tests PRIVATE QUICKQR_TEST_BUILD QUICKQR_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(qr_ui_tests PRIVATE qr_core comctl32 shell32 shlwapi ole32 dwmapi comdlg32 psapi)
target_compile_options(qr_ui_tests PRIVATE /utf-8 /W4)
target_link_options(qr_ui_tests PRIVATE /MANIFEST:NO)
